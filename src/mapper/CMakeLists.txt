cmake_minimum_required(VERSION 3.8)
project(mapper)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(uav_navigation_msgs REQUIRED)
find_package(voxel_grid_core REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
add_executable(static_mapper src/mapper_node.cpp)
target_compile_features(static_mapper PRIVATE cxx_std_17)
target_compile_options(static_mapper PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(static_mapper rclcpp sensor_msgs pcl_conversions uav_navigation_msgs)
target_include_directories(static_mapper PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(static_mapper ${PCL_LIBRARIES} voxel_grid_core::voxel_grid_core)
install(TARGETS static_mapper DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
