cmake_minimum_required(VERSION 3.8)
project(astar_planner)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(uav_navigation_msgs REQUIRED)
find_package(voxel_grid_core REQUIRED)
add_library(planner_core INTERFACE)
target_compile_features(planner_core INTERFACE cxx_std_17)
target_include_directories(planner_core INTERFACE include)
target_link_libraries(planner_core INTERFACE ${uav_navigation_msgs_TARGETS})
target_link_libraries(planner_core INTERFACE voxel_grid_core::voxel_grid_core)
add_executable(astar_planner_node src/planner_node.cpp)
target_compile_options(astar_planner_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(astar_planner_node rclcpp rclcpp_action nav_msgs geometry_msgs uav_navigation_msgs)
target_link_libraries(astar_planner_node planner_core)
install(TARGETS astar_planner_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_search test/test_search.cpp)
  ament_target_dependencies(test_search uav_navigation_msgs)
  target_link_libraries(test_search planner_core)
endif()
ament_package()
